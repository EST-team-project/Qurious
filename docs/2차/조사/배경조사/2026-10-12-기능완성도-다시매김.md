**부록 A 다시 채우기** — 보고서가 마지막으로 바뀐 `b95e945` 의 점검표로 돌리면 그 판의 부록 A 와 글자까지 같다(확인). 표시 주석 사이(`<!-- completion_table:start -->` ~ `end`)를 이 출력으로 바꾼다.

```python
# python -X utf8 이_코드.py docs/시험/대장/기능완성도-점검표.tsv
import csv, io, re, sys
DELTA_RE = re.compile(r"(?:^| · )([+-]\d+) ")   # TC-CP-04 와 같은 규칙
rows = list(csv.DictReader(io.StringIO(open(sys.argv[1], encoding="utf-8", newline="").read(), newline=""), delimiter="\t"))
out = ["| # | 화면 | 화면 키 | 메뉴 | 주담당 | 강사님 | 점수 | 이름표 | 매긴 법 | 100% 까지 |",
       "|--:|---|---|---|---|:-:|:-:|---|---|---|"]
for i, r in enumerate(rows, 1):
    if r["방법"] == "배점표":
        how = "배점표 " + "/".join(r[c] for c in ("UI", "서버", "연동", "시험"))
    else:
        d = DELTA_RE.findall(r["고침"])
        how = ("고침 " + " ".join(d)) if d else "그대로"
    out.append(f"| {i} | {r['화면명']} | `{r['화면키']}` | {r['메뉴']} | {r['주담당']} | "
               f"{r['강사님'] or '미평가'} | {r['점수']} | {r['이름표']} | {how} | {r['100%까지']} |")
sys.stdout.reconfigure(encoding="utf-8")
print("\n".join(out))
```
